cmake_minimum_required(VERSION 3.8)
project(native_library)
set(CMAKE_CXX_STANDARD 17)

set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY build/lib)
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY build/lib)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY build/lib)

if("${CMAKE_CXX_COMPILER_ID}" STREQUAL "GNU")
    if (CMAKE_CXX_COMPILER_VERSION VERSION_LESS 6.0)
        message(FATAL_ERROR "GCC version must be at least 6.0!")
    endif()
	set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -O2 -s -DNDEBUG")
elseif ("${CMAKE_CXX_COMPILER_ID}" STREQUAL "MSVC")
    set(CMAKE_LIBRARY_OUTPUT_DIRECTORY build/lib)
    set(CMAKE_LIBRARY_OUTPUT_DIRECTORY_DEBUG build/lib)
    set(CMAKE_LIBRARY_OUTPUT_DIRECTORY_RELEASE build/lib)
    set(CMAKE_RUNTIME_OUTPUT_DIRECTORY build/lib)
    set(CMAKE_RUNTIME_OUTPUT_DIRECTORY_DEBUG build/lib)
    set(CMAKE_RUNTIME_OUTPUT_DIRECTORY_RELEASE build/lib)
	set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} /MP8 /wd4102 /wd4101 /EHsc")
elseif ("${CMAKE_CXX_COMPILER_ID}" STREQUAL "Clang")
elseif ("${CMAKE_CXX_COMPILER_ID}" STREQUAL "AppleClang")
else()
    message(AUTHOR_WARNING "You are using an unsupported compiler! Compilation has only been tested with MSVC, GNU, Clang and AppleClang.")
endif()

find_package(JNI REQUIRED)
include_directories(${JNI_INCLUDE_DIRS})

set(CLASS_FILES output/dev_sakura_L3MonKe_I_0.hpp output/dev_sakura_L3MonKe_I_0.cpp output/dev_sakura_L3MonKe_O_1.hpp output/dev_sakura_L3MonKe_O_1.cpp output/dev_sakura_L3MonKe_S_2.hpp output/dev_sakura_L3MonKe_S_2.cpp output/dev_sakura_L3MonKe_W_3.hpp output/dev_sakura_L3MonKe_W_3.cpp output/dev_sakura_L3MonKe_d_4.hpp output/dev_sakura_L3MonKe_d_4.cpp output/dev_sakura_L3MonKe_f_5.hpp output/dev_sakura_L3MonKe_f_5.cpp output/dev_sakura_L3MonKe_h_6.hpp output/dev_sakura_L3MonKe_h_6.cpp output/dev_sakura_L3MonKe_r_7.hpp output/dev_sakura_L3MonKe_r_7.cpp output/dev_sakura_client_L3MonKe_Tq_8.hpp output/dev_sakura_client_L3MonKe_Tq_8.cpp output/data_native0_hidden_Hidden0.hpp output/data_native0_hidden_Hidden0.cpp)
set(MAIN_FILES native_jvm.hpp native_jvm.cpp native_jvm_output.hpp native_jvm_output.cpp string_pool.hpp string_pool.cpp)
add_definitions(-DUSE_HOTSPOT=1)

add_library(native_library SHARED ${CLASS_FILES} ${MAIN_FILES})
